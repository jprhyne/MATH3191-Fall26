        T\left(\bMat{1\\2\\3}\right) = \bMat{0\\6\\2}, T\left(\bMat{1 \\ 1 \\ 1}\right) = \bMat{1\\3\\1}, T\left(\bMat{0\\2\\5}\right) = \bMat{-3 \\ 7 \\ 3}
